import logging import re from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup from telegram.ext import Application, CommandHandler, MessageHandler, filters, ContextTypes # ---------- CONFIG ---------- BOT_TOKEN = "YOUR_BOT_TOKEN_HERE" # Replace with your bot token from @BotFather IMAGE_FILE = "welcome.jpg" # Put your image file in the same folder # ---------- LOGGING ---------- logging.basicConfig(format="%(asctime)s - %(name)s - %(levelname)s - %(message)s", level=logging.INFO) # ---------- WELCOME MESSAGE ---------- WELCOME_TEXT = ( "✅ VIP has increased to 3.5% + 3📌\n\n" "🪙 REGISTER HERE ⏩⏩\n" "https://app-web.mobiuspe-app.com/regist?code=earnmoney426\n\n" "✅ We offer team leader salaries and up to 0.6% team commission. " "Please contact us to apply for a team leader position. 🛒\n\n" "Official channel link ⭐️\n" "https://t.me/mobiuspayofficial1\n\n" "Contact support ⭐️ @puya1521" ) # ---------- HELPER FUNCTIONS ---------- def clean_text(text: str) -> str: # Remove extra spaces, tabs, multiple newlines text = re.sub(r"\s+", " ", text).strip() return text def strip_trackers(text: str) -> str: # Remove common UTM/tracking parameters from URLs return re.sub(r"[?&](utm_[^&]+|ref|code|source|campaign)=[^&]*", "", text) def count_ad_chars(text: str) -> dict: with_spaces = len(text) without_spaces = len(text.replace(" ", "")) return {"with": with_spaces, "without": without_spaces} # ---------- /start COMMAND ---------- async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): user = update.effective_user caption = f"👋 G'day {user.first_name}!\n\n{WELCOME_TEXT}\n\n---\n🧰 *CopyBot Functions:*\n• Clean text spaces\n• Strip link trackers\n• Count ad characters\n\nSend me any text or URL, and I'll process it!" try: with open(IMAGE_FILE, "rb") as img: await update.message.reply_photo( photo=img, caption=caption, parse_mode="Markdown" ) except FileNotFoundError: # Fallback if image missing await update.message.reply_text( f"👋 G'day {user.first_name}!\n\n{WELCOME_TEXT}\n\n---\n🧰 Send me text or a URL to clean/count.", parse_mode="Markdown" ) # ---------- TEXT PROCESSING ---------- async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE): raw = update.message.text # Step 1: Clean spaces cleaned = clean_text(raw) # Step 2: Strip trackers stripped = strip_trackers(cleaned) # Step 3: Count characters counts = count_ad_chars(stripped) response = ( f"📝 *Cleaned & Stripped:*\n`{stripped}`\n\n" f"📊 *Character count:*\n" f"• With spaces: {counts['with']}\n" f"• Without spaces: {counts['without']}" ) await update.message.reply_text(response, parse_mode="Markdown") # ---------- MAIN ---------- def main(): app = Application.builder().token(BOT_TOKEN).build() app.add_handler(CommandHandler("start", start)) app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_text)) print("🤖 Bot is running... Press Ctrl+C to stop.") app.run_polling(allowed_updates=Update.ALL_TYPES) if __name__ == "__main__": main()
